set -euo pipefail

start=$(date +%s)
date
echo "Using NSLOTS=${NSLOTS:-NA}"

source /old_Users/jayelazuno/miniforge3/etc/profile.d/conda.sh
conda activate tnseq

YTAB="/Users/jayelazuno/workspace/ytab"
DATA_DIR="/Users/jayelazuno/workspace/HermesTnseq/data"
BOWTIE2_INDEX="/Users/jayelazuno/workspace/HermesTnseq/transposon-pipeline/dependencies/glabrata/reference_genome/GCA_014217725.1_ASM1421772v1_genomic"

RUN_DIR="$YTAB/output/smoketests/mapfastq"
EXPORT_DIR="$YTAB/output/exports/smoke_test_v1"

mkdir -p "$RUN_DIR"
mkdir -p "$EXPORT_DIR"

THREADS=56
export PYTHONPATH="$YTAB/src"

FASTQS=(
  "yH298-H2O2-treated-facs-pool1.fastq.gz"
)

echo "YTAB repo:        $YTAB"
echo "Input FASTQ dir:  $DATA_DIR"
echo "Run dir:          $RUN_DIR"
echo "Export dir:       $EXPORT_DIR"
echo "Index prefix:     $BOWTIE2_INDEX"
echo

for fq_base in "${FASTQS[@]}"; do
  fq="$DATA_DIR/$fq_base"
  if [[ ! -s "$fq" ]]; then
    echo "ERROR: missing input FASTQ: $fq"
    exit 1
  fi

  sample="${fq_base%.fastq.gz}"
  sample_out="$RUN_DIR/$sample"
  mkdir -p "$sample_out"

  echo "[$(date)] sample=$sample"
  echo "  input=$fq"
  echo "  output=$sample_out"

  python3 "$YTAB/src/ytab/mapping/MapFastq.py" \
    -i "$fq" \
    -o "$sample_out" \
    -x "$BOWTIE2_INDEX" \
    -t "$THREADS"

  cp "$sample_out/${sample}.mapping_stats.csv" "$EXPORT_DIR/${sample}.mapping_stats.csv"
  echo
done

date
end=$(date +%s)
runtime=$((end-start))
echo "runtime=$runtime secs"

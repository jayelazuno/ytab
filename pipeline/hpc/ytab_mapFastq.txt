#!/bin/bash
#$ -S /bin/bash
#$ -cwd
#$ -N YTAB_MapFastq
#$ -q UI
#$ -pe smp 8
#$ -M jayelazuno@uiowa.edu
#$ -m e
#$ -j y
#$ -o YTAB_MapFastq.log

## Personal University of Iowa Argon / SGE alignment job, not a portable scheduler interface.
## qsub -v PROJECT_ID=H2O2_screen_v1 pipeline/hpc/ytab_mapFastq.txt
## qsub -v PROJECT_CONFIG=/absolute/path/project.yaml pipeline/hpc/ytab_mapFastq.txt
## qsub -v PROJECT_ID=Gale_2020,JUNCTION_MATE=R1 pipeline/hpc/ytab_mapFastq.txt
## One task per sample, whole project as a single job (tasks past the per-user limit queue instead of being rejected):
## qsub -t 1-33 -v PROJECT_ID=Patterson_2018_lipolytica pipeline/hpc/ytab_mapFastq.txt
## Optional: FORCE=1, DRY_RUN=1, SAMPLES=one_sample, REQUIRE_MATE_ALIGNMENT=1, SAMPLE_LIST=file_of_sample_names
## Change CPU allocation with qsub -pe smp 16; the CLI uses NSLOTS.

set -euo pipefail
start=$(date +%s)
trap 'rc=$?; echo "Finished: $(date); exit code=$rc; elapsed=$(( $(date +%s) - start )) seconds"' EXIT

YTAB="${YTAB:-/Users/jayelazuno/workspace/ytab}"
YTAB_ENV_MODE="${YTAB_ENV_MODE:-conda}"
YTAB_CONDA_SH="${YTAB_CONDA_SH:-/old_Users/jayelazuno/miniforge3/etc/profile.d/conda.sh}"
YTAB_CONDA_ENV="${YTAB_CONDA_ENV:-tnseq}"
PYTHON_BIN="${PYTHON_BIN:-python}"
PROJECT_ID="${PROJECT_ID:-}"
PROJECT_CONFIG="${PROJECT_CONFIG:-}"
JUNCTION_MATE="${JUNCTION_MATE:-}"
REQUIRE_MATE_ALIGNMENT="${REQUIRE_MATE_ALIGNMENT:-}"
FORCE="${FORCE:-0}"
DRY_RUN="${DRY_RUN:-0}"
KEEP_GOING="${KEEP_GOING:-1}"
SCRATCH_ROOT="${SCRATCH_ROOT:-/nfsscratch/jayelazuno/workspace/ytab_alignment}"

if [[ ! -d "$YTAB" ]]; then
    echo "ERROR: YTAB checkout not found: $YTAB" >&2
    exit 2
fi
YTAB="$(cd "$YTAB" && pwd -P)"

if [[ -z "$PROJECT_CONFIG" ]]; then
    if [[ -z "$PROJECT_ID" || ! "$PROJECT_ID" =~ ^[A-Za-z0-9][A-Za-z0-9_.-]*$ ]]; then
        echo "ERROR: Supply PROJECT_CONFIG or a valid PROJECT_ID with qsub -v." >&2
        exit 2
    fi
    PROJECT_CONFIG="$YTAB/output/projects/$PROJECT_ID/config/project.yaml"
elif [[ "$PROJECT_CONFIG" != /* ]]; then
    PROJECT_CONFIG="$PWD/$PROJECT_CONFIG"
fi

if [[ ! -s "$PROJECT_CONFIG" || ! -f "$YTAB/scripts/local/ytab.py" ]]; then
    echo "ERROR: Project configuration or unified CLI is missing." >&2
    echo "Project: $PROJECT_CONFIG" >&2
    exit 2
fi

THREADS="${NSLOTS:-}"
if [[ ! "$THREADS" =~ ^[1-9][0-9]*$ ]] || (( THREADS < 2 )); then
    echo "ERROR: Submit with qsub and allocate at least two smp slots." >&2
    exit 2
fi
for value in "$FORCE" "$DRY_RUN" "$KEEP_GOING"; do
    if [[ "$value" != 0 && "$value" != 1 ]]; then
        echo "ERROR: FORCE, DRY_RUN and KEEP_GOING must be 0 or 1." >&2
        exit 2
    fi
done
case "$JUNCTION_MATE" in ""|R1|R2) ;; *) echo "ERROR: JUNCTION_MATE must be R1 or R2." >&2; exit 2 ;; esac
case "$REQUIRE_MATE_ALIGNMENT" in ""|0|1) ;; *) echo "ERROR: REQUIRE_MATE_ALIGNMENT must be 0 or 1." >&2; exit 2 ;; esac

case "$YTAB_ENV_MODE" in
    conda)
        if [[ ! -s "$YTAB_CONDA_SH" ]]; then
            echo "ERROR: Conda initialization script missing: $YTAB_CONDA_SH" >&2
            exit 2
        fi
        source "$YTAB_CONDA_SH"
        conda activate "$YTAB_CONDA_ENV"
        ;;
    modules)
        if ! type module >/dev/null 2>&1; then
            if [[ -f /etc/profile.d/modules.sh ]]; then
                source /etc/profile.d/modules.sh
            elif [[ -f /usr/share/lmod/lmod/init/bash ]]; then
                source /usr/share/lmod/lmod/init/bash
            else
                echo "ERROR: Environment modules unavailable in this batch shell." >&2
                exit 2
            fi
        fi
        if [[ -z "${YTAB_MODULES:-}" ]]; then
            echo "ERROR: Set YTAB_MODULES to the space-separated module names for this installation." >&2
            exit 2
        fi
        read -r -a requested_modules <<< "$YTAB_MODULES"
        module load "${requested_modules[@]}"
        module list 2>&1
        ;;
    *) echo "ERROR: YTAB_ENV_MODE must be conda or modules." >&2; exit 2 ;;
esac

for tool in "$PYTHON_BIN" bowtie2 samtools; do
    command -v "$tool" >/dev/null || {
        echo "ERROR: Required executable missing: $tool" >&2
        exit 2
    }
done
"$PYTHON_BIN" -c 'import cutadapt, dnaio, Bio, pysam, yaml, pandas'

mkdir -p "$SCRATCH_ROOT"
scratch_job=$(mktemp -d "$SCRATCH_ROOT/mapfastq.${JOB_ID:-manual}.XXXXXX")
export TMPDIR="$scratch_job"
exec > >(tee -a "$scratch_job/alignment.log") 2>&1

echo "Started: $(date)"
echo "Host: $(hostname)"
echo "Job: ${JOB_ID:-manual}; NSLOTS=$THREADS"
echo "YTAB: $YTAB"
echo "Project: $PROJECT_CONFIG"
echo "Temporary files/job log: $scratch_job"
echo "Junction mate: ${JUNCTION_MATE:-project settings; default R1}"
echo "Mate agreement: ${REQUIRE_MATE_ALIGNMENT:-project settings; default no}"
df -h "$SCRATCH_ROOT"
"$PYTHON_BIN" --version
bowtie2 --version
samtools --version

## Array mode: with qsub -t, each task maps one sample, so a whole project goes in as one job and the tasks
## beyond the per-user limit wait in the queue instead of being rejected at submit time.
##   qsub -t 1-33 -v PROJECT_ID=Patterson_2018_lipolytica pipeline/hpc/ytab_mapFastq.txt
## The task's sample is the SGE_TASK_ID'th included sample of the project, in sample-sheet order, so nothing has
## to be prepared by hand. SAMPLE_LIST still overrides that with one sample name per line, for mapping a subset.
if [[ "${SGE_TASK_ID:-undefined}" =~ ^[1-9][0-9]*$ ]]; then
    if [[ -n "${SAMPLE_LIST:-}" ]]; then
        [[ "$SAMPLE_LIST" = /* ]] || SAMPLE_LIST="$PWD/$SAMPLE_LIST"
        [[ -s "$SAMPLE_LIST" ]] || { echo "ERROR: SAMPLE_LIST not found or empty: $SAMPLE_LIST" >&2; exit 2; }
        task_samples="$SAMPLE_LIST"
        source_description="$SAMPLE_LIST"
    else
        # The project's own included samples, read through the pipeline's loader so the array maps exactly what
        # a whole-project run would map, in the same order.
        task_samples="$scratch_job/array_samples.txt"
        (cd "$YTAB" && PYTHONPATH="$YTAB/src" "$PYTHON_BIN" -c '
import sys
from pathlib import Path
from ytab.pipeline.mapfastq_runner import get_included_samples, load_project_for_mapping
for sample in get_included_samples(load_project_for_mapping(Path(sys.argv[1]))):
    print(sample["sample"])
' "$PROJECT_CONFIG") > "$task_samples"
        [[ -s "$task_samples" ]] || { echo "ERROR: No included samples in $PROJECT_CONFIG" >&2; exit 2; }
        source_description="$PROJECT_CONFIG"
    fi
    SAMPLES="$(sed -n "${SGE_TASK_ID}p" "$task_samples" | tr -d '[:space:]')"
    total="$(grep -c '[^[:space:]]' "$task_samples")"
    [[ -n "$SAMPLES" ]] || {
        echo "ERROR: Task $SGE_TASK_ID has no sample; $source_description lists $total. Submit with qsub -t 1-$total." >&2
        exit 2
    }
    if [[ "${SGE_TASK_LAST:-$total}" != "$total" ]]; then
        echo "WARNING: the array range ends at ${SGE_TASK_LAST}, but $source_description lists $total samples." >&2
        echo "         Samples outside 1-${SGE_TASK_LAST} will not be mapped; resubmit with qsub -t 1-$total." >&2
    fi
    echo "Array task $SGE_TASK_ID of $total: $SAMPLES"
elif [[ -n "${SAMPLE_LIST:-}" ]]; then
    echo "ERROR: SAMPLE_LIST requires an array job; submit with qsub -t 1-N." >&2
    exit 2
fi

args=("$YTAB/scripts/local/ytab.py" mapfastq
      --project-config "$PROJECT_CONFIG" --threads "$THREADS")
[[ -n "${SAMPLES:-}" ]] && args+=(--samples "$SAMPLES")
[[ -n "$JUNCTION_MATE" ]] && args+=(--junction-mate "$JUNCTION_MATE")
[[ "$REQUIRE_MATE_ALIGNMENT" == 1 ]] && args+=(--require-mate-alignment)
[[ "$REQUIRE_MATE_ALIGNMENT" == 0 ]] && args+=(--no-require-mate-alignment)
[[ "$FORCE" == 1 ]] && args+=(--force)
[[ "$DRY_RUN" == 1 ]] && args+=(--dry-run)
[[ "$KEEP_GOING" == 1 ]] && args+=(--keep-going)

printf 'Command: '; printf '%q ' "$PYTHON_BIN" "${args[@]}"; printf '\n'
cd "$YTAB"
"$PYTHON_BIN" "${args[@]}"

echo "SUCCESS: alignment stage completed (or previewed when DRY_RUN=1)."
echo "Final BAMs, mapping QC and manifests use the existing project configuration's output paths."
echo "No hit-file, summary, normalization, classifier or fitness jobs were submitted."

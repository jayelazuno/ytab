
#!/bin/bash
#$ -S /bin/bash
#$ -cwd
#$ -N YTAB_SummaryTable
#$ -q UI
#$ -pe smp 56
#$ -M jayelazuno@uiowa.edu
#$ -m e
#$ -j y
#$ -o YTAB_SummaryTable.log

set -euo pipefail

start=$(date +%s)
date
echo "Using NSLOTS=${NSLOTS:-NA}"

# -----------------------
# Activate tnseq conda env
# -----------------------
source /old_Users/jayelazuno/miniforge3/etc/profile.d/conda.sh
conda activate tnseq

# -----------------------
# Project paths
# -----------------------
YTAB="/Users/jayelazuno/workspace/ytab"
SCRIPT="$YTAB/src/ytab/summary/SummaryTable.py"

# Input hit files from YTAB CreateHitFile smoke test
HITS_ROOT="$YTAB/output/smoketests/create_hit_file"

# Stage output for SummaryTable
OUT_ROOT="$YTAB/output/smoketests/summary"

# App-facing export root
EXPORT_ROOT="$YTAB/output/exports/smoke_test_v1/qc/summary"

mkdir -p "$OUT_ROOT"
mkdir -p "$EXPORT_ROOT"

THREADS=${NSLOTS:-56}
echo "THREADS(for SummaryTable)=$THREADS"

echo "YTAB repo:     $YTAB"
echo "Hits root:     $HITS_ROOT"
echo "Out root:      $OUT_ROOT"
echo "Export root:   $EXPORT_ROOT"
echo "Script:        $SCRIPT"
echo

export PYTHONPATH="$YTAB/src"

# -----------------------
# Smoke-test sample(s)
# -----------------------
SAMPLES=(
  "yH298-parent-pool1"
)

for sample in "${SAMPLES[@]}"; do
  in_dir="$HITS_ROOT/$sample"
  out="$OUT_ROOT/$sample"
  sample_export_dir="$EXPORT_ROOT/$sample"

  if [[ ! -d "$in_dir" ]]; then
    echo "ERROR: missing input directory: $in_dir" >&2
    exit 1
  fi

  hits_files=( "$in_dir"/*_hits.txt )
  if [[ ${#hits_files[@]} -eq 0 ]]; then
    echo "ERROR: no *_hits.txt found in: $in_dir" >&2
    exit 1
  fi

  mkdir -p "$out"
  mkdir -p "$sample_export_dir"

  echo "[$(date)] sample=$sample"
  echo "  input=$in_dir"
  echo "  output=$out"
  echo "  export=$sample_export_dir"

  python3 "$SCRIPT" \
    -i "$in_dir" \
    -o "$out" \
    --hit-glob "*_hits.txt" \
    --hit-format hit_table \
    --feature-db glabrata \
    -f 1 \
    --neighborhood-window-size 10000 \
    --bin-size 10000 \
    -t "$THREADS" \
    --overwrite

  # -----------------------
  # Copy app-facing summary assets
  # -----------------------
  for f in \
    "$out/stats.csv" \
    "$out/${sample}.all_hits.csv" \
    "$out/${sample}.feature_table.RDF_1.csv" \
    "$out/${sample}_analysis.csv" \
    "$out/binned_hits.RDF_1.csv" \
    "$out/hit_summary.RDF_1.csv" \
    "$out/insertion_vs_neighborhood_correlations.txt" \
    "$out/outlier_stats.txt"
  do
    if [[ -s "$f" ]]; then
      cp -f "$f" "$sample_export_dir/"
    else
      echo "[WARN] expected file missing: $f"
    fi
  done

  echo
done

date
end=$(date +%s)
runtime=$((end-start))
echo "runtime=$runtime secs"
echo "Done."
